import { performance } from "node:perf_hooks";
const m = await import("./overlay.min.mjs");
const cat = m.fullCatalog();
const ui = cat.filter((e) => e.namespace === "ui"), email = cat.filter((e) => e.namespace === "email");
console.log("catalogue entries", cat.length, "ui", ui.length, "email", email.length, "templates", cat.filter((e) => e.kind === "template").length);
const rows = cat.map((e) => ({ key: e.key, text: e.source }));
console.log("rows as JSON bytes (one language, English lengths)", JSON.stringify(rows).length);
const en = m.t("en"), em = m.emailText("en");
function once() {
  const uiMap = new Map(), emailMap = new Map();
  for (const row of rows) (row.key.startsWith("email:") ? emailMap : uiMap).set(row.key, row.text);
  const a = m.overlayMessages("ui", en, uiMap, "de");
  const b = m.overlayMessages("email", em, emailMap, "de");
  return [a, b, Object.fromEntries(uiMap)];
}
const times = [];
const keep = [];
if (global.gc) global.gc();
const h0 = process.memoryUsage().heapUsed;
for (let i = 0; i < 20; i++) { const t0 = performance.now(); keep.push(once()); times.push(performance.now() - t0); }
if (global.gc) global.gc();
const h1 = process.memoryUsage().heapUsed;
console.log("overlay ms: first", times[0].toFixed(2), "| 2nd-5th", times.slice(1, 5).map((x) => x.toFixed(2)).join(" "), "| mean of 20", (times.reduce((a, b) => a + b) / 20).toFixed(2));
console.log("heap kept per language (KB), 20 languages held:", ((h1 - h0) / 20 / 1024).toFixed(0));
const browserKeys = new Set(m.catalogOf("ui", m.enBrowser).map((e) => e.key));
const returnsKeys = new Set(m.catalogOf("ui", { returns: m.enReturns }).map((e) => e.key));
const uiObj = Object.fromEntries(ui.map((e) => [e.key, e.source]));
const pick = (set) => Object.fromEntries(Object.entries(uiObj).filter(([k]) => set.has(k)));
console.log("ui keys", ui.length, "bytes", JSON.stringify(uiObj).length, "| browser keys", browserKeys.size, "in catalogue", Object.keys(pick(browserKeys)).length, "bytes", JSON.stringify(pick(browserKeys)).length, "| returns keys", returnsKeys.size, "bytes", JSON.stringify(pick(returnsKeys)).length);

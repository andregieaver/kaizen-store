// Static value-import graph from an entry file over src (ignores `import type` and type-only named imports). Usage: node graph.mjs <entry> [--why target]
import fs from "node:fs"; import path from "node:path";
const root = process.cwd();
const exts = [".ts", ".tsx", ".js", ".mjs", ".json", "/index.ts", "/index.tsx"];
function resolve(from, spec) {
  let base;
  if (spec.startsWith("@/")) base = path.join(root, "src", spec.slice(2));
  else if (spec.startsWith(".")) base = path.resolve(path.dirname(from), spec);
  else return null;
  if (fs.existsSync(base) && fs.statSync(base).isFile()) return base;
  for (const e of exts) if (fs.existsSync(base + e)) return base + e;
  return null;
}
const importRe = /^\s*(import|export)\s+(type\s+)?([^;]*?)\s+from\s+["']([^"']+)["']|^\s*import\s+["']([^"']+)["']/gm;
function imports(file) {
  if (file.endsWith(".json")) return { local: [], ext: [] };
  const text = fs.readFileSync(file, "utf8");
  const local = [], ext = [];
  let m; importRe.lastIndex = 0;
  while ((m = importRe.exec(text))) {
    if (m[2]) continue; // import type
    const spec = m[4] ?? m[5];
    const clause = m[3] ?? "";
    // `import { type A, type B } from` is erased
    const braces = clause.match(/^\{([\s\S]*)\}$/);
    if (braces && braces[1].split(",").map((s) => s.trim()).filter(Boolean).every((s) => s.startsWith("type "))) continue;
    const r = resolve(file, spec);
    if (r) local.push(r); else if (!spec.startsWith(".") && !spec.startsWith("@/")) ext.push(spec);
  }
  return { local, ext };
}
const entry = path.resolve(process.argv[2]); const CUT = new Set((process.env.CUT || "").split(",").filter(Boolean));
const seen = new Map([[entry, null]]); const queue = [entry]; const pkgs = new Set();
while (queue.length) { const f = queue.shift(); const { local, ext } = imports(f); ext.forEach((e) => pkgs.add(e)); for (const l of local) if (!CUT.has(path.relative(root, f) + ">" + path.relative(root, l)) && !CUT.has("*>" + path.relative(root, l)) && !seen.has(l)) { seen.set(l, f); queue.push(l); } }
let bytes = 0; const group = {};
for (const f of seen.keys()) { const s = fs.statSync(f).size; bytes += s; const rel = path.relative(root, f); const g = rel.split("/").slice(0, 2).join("/"); group[g] = group[g] || [0, 0]; group[g][0]++; group[g][1] += s; }
console.log("reachable modules", seen.size, "source bytes", bytes);
for (const [g, [n, b]] of Object.entries(group).sort((a, b) => b[1][1] - a[1][1])) console.log(" ", g.padEnd(18), String(n).padStart(5), String(b).padStart(10));
console.log("packages:", [...pkgs].sort().join(", "));
const whyIdx = process.argv.indexOf("--why");
if (whyIdx > 0) for (const target of process.argv.slice(whyIdx + 1)) { let t = [...seen.keys()].find((k) => k.endsWith(target)); const chain = []; while (t) { chain.push(path.relative(root, t)); t = seen.get(t); } console.log("why", target, ":\n   " + chain.reverse().join("\n   -> ")); }

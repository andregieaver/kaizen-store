import { createRequire } from "node:module";
import { readFileSync, writeFileSync } from "node:fs";
import { gzipSync, brotliCompressSync } from "node:zlib";
import path from "node:path";
const req = createRequire("/home/user/kaizen-store/node_modules/.pnpm/@tailwindcss+postcss@4.3.3/node_modules/@tailwindcss/postcss/");
const { compile, optimize } = req("@tailwindcss/node");
const { Scanner } = req("@tailwindcss/oxide");
const R = "/home/user/kaizen-store";
const real = readFileSync(R + "/src/app/globals.css", "utf8");
const HEAD = `@source "../";\n@source not "../data";\n@source "../../supabase/migrations/*.sql";`;
if (!real.includes(HEAD)) throw new Error("head not found");
const NOT_NOISE = `@source not "../**/*.test.ts";\n@source not "../**/*.test.tsx";\n@source not "../test";\n@source not "../lib/messages";\n@source not "../lib/admin-messages";\n@source not "../db";`;
const SHARED = ["action-form.tsx","admin-text-context.tsx","admin-text.tsx","colour-field.tsx","colour-library.tsx","css-code-editor.tsx","design-cards.tsx","modal.tsx","rich-text-editor.tsx","text-styles.tsx","work/print-button.tsx"].map(f => `@source "../components/admin/${f}";`).join("\n");
const variants = {
  today: real,
  lean: real.replace(HEAD, `@source "../";\n@source not "../data";\n${NOT_NOISE}`),
  site: real.replace(HEAD, `@source "../";\n@source not "../data";\n@source not "./admin";\n@source not "../components/admin";\n${NOT_NOISE}\n${SHARED}`),
  siteNoShared: real.replace(HEAD, `@source "../";\n@source not "../data";\n@source not "./admin";\n@source not "../components/admin";\n${NOT_NOISE}`),
};
const results = {};
for (const [name, css] of Object.entries(variants)) {
  const file = R + "/src/app/globals.css";
  const compiler = await compile(css, { base: path.dirname(file), from: file, onDependency: () => {} });
  const sources = compiler.sources.map(s => ({ base: s.base, pattern: s.pattern, negated: s.negated }));
  const t0 = Date.now();
  const scanner = new Scanner({ sources });
  const cands = scanner.scan();
  const scanMs = Date.now() - t0;
  const out = compiler.build(cands);
  let min = out;
  try { min = optimize(out, { minify: true }).code; } catch (e) { console.log("optimize failed", e.message); }
  results[name] = min;
  const adminFiles = scanner.files.filter(f => f.includes("/components/admin/") || f.includes("/app/admin/")).length;
  console.log(name.padEnd(13), "files", scanner.files.length, "(admin", adminFiles + ")", "candidates", cands.length, "scan ms", scanMs, "| css raw", Buffer.byteLength(out), "min", Buffer.byteLength(min), "gz9", gzipSync(min, { level: 9 }).length, "br", brotliCompressSync(min).length, "rssMB", Math.round(process.memoryUsage().rss / 1e6));
  writeFileSync(`out-${name}.css`, min);
}

import { createRequire } from "node:module";
import { readFileSync, writeFileSync } from "node:fs";
import { gzipSync, brotliCompressSync } from "node:zlib";
import path from "node:path";
const req = createRequire("/home/user/kaizen-store/node_modules/.pnpm/@tailwindcss+postcss@4.3.3/node_modules/@tailwindcss/postcss/");
const { compile, optimize } = req("@tailwindcss/node");
const { Scanner } = req("@tailwindcss/oxide");
const R = "/home/user/kaizen-store";
const real = readFileSync(R + "/src/app/globals.css", "utf8");
const lines = real.split("\n");
// builder-only: lines 362..434 and 630..656 (1-indexed, inclusive)
const kept = lines.filter((_, i) => { const n = i + 1; return !((n >= 362 && n <= 434) || (n >= 630 && n <= 656)); }).join("\n");
const cut = lines.filter((_, i) => { const n = i + 1; return (n >= 362 && n <= 434) || (n >= 630 && n <= 656); }).join("\n");
console.log("builder-only source lines", cut.split("\n").length, "source bytes", Buffer.byteLength(cut), "mentions outside data-builder/data-resize:", (cut.match(/^[^\s/*].*\{/gm) || []).filter(l => !/data-builder|data-resize|@media/.test(l)));
const HEAD = `@source "../";\n@source not "../data";\n@source "../../supabase/migrations/*.sql";`;
const NOT_NOISE = `@source not "../**/*.test.ts";\n@source not "../**/*.test.tsx";\n@source not "../test";\n@source not "../lib/messages";\n@source not "../lib/admin-messages";\n@source not "../db";`;
const SHARED = ["action-form.tsx","admin-text-context.tsx","admin-text.tsx","colour-field.tsx","colour-library.tsx","css-code-editor.tsx","design-cards.tsx","modal.tsx","rich-text-editor.tsx","text-styles.tsx","work/print-button.tsx"].map(f => `@source "../components/admin/${f}";`).join("\n");
const siteHead = `@source "../";\n@source not "../data";\n@source not "./admin";\n@source not "../components/admin";\n${NOT_NOISE}\n${SHARED}`;
const variants = { siteFinal: kept.replace(HEAD, siteHead), adminFinal: real.replace(HEAD, `@source "../";\n@source not "../data";\n${NOT_NOISE}`) };
let rss0 = process.memoryUsage().rss;
for (const [name, css] of Object.entries(variants)) {
  const file = R + "/src/app/globals.css";
  const compiler = await compile(css, { base: path.dirname(file), from: file, onDependency: () => {} });
  const scanner = new Scanner({ sources: compiler.sources.map(s => ({ base: s.base, pattern: s.pattern, negated: s.negated })) });
  const t0 = Date.now(); const cands = scanner.scan(); const ms = Date.now() - t0;
  const min = optimize(compiler.build(cands), { minify: true }).code;
  console.log(name.padEnd(11), "files", scanner.files.length, "candidates", cands.length, "scan ms", ms, "| min", Buffer.byteLength(min), "gz9", gzipSync(min, { level: 9 }).length, "br", brotliCompressSync(min).length, "| rss MB", Math.round(process.memoryUsage().rss / 1e6));
  writeFileSync(`out-${name}.css`, min);
}

import { readFileSync, statSync, readdirSync } from "node:fs";
import { join, resolve, dirname } from "node:path";
const root = "/home/user/kaizen-store/.next/server/app/api/cron";
for (const name of readdirSync(root)) {
  const file = join(root, name, "route.js");
  let src;
  try { src = readFileSync(file, "utf8"); } catch { continue; }
  const chunks = [...src.matchAll(/R\.c\("([^"]+)"\)/g)].map((m) => m[1]);
  let bytes = 0;
  for (const c of chunks) { try { bytes += statSync(join("/home/user/kaizen-store/.next", c)).size; } catch {} }
  const nft = JSON.parse(readFileSync(file + ".nft.json", "utf8"));
  let traced = 0;
  for (const f of nft.files) { try { const s = statSync(resolve(dirname(file), f)); if (s.isFile()) traced += s.size; } catch {} }
  console.log(`${name}\tchunks=${chunks.length}\tjs_at_start=${(bytes / 1e6).toFixed(1)}MB\ttraced=${(traced / 1e6).toFixed(1)}MB\tfiles=${nft.files.length}`);
}

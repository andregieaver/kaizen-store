import fs from "node:fs";
const d = JSON.parse(fs.readFileSync("../../m/routes.json"));
const names = process.argv.slice(2);
for (const p of d) {
  if (names.length && !names.includes(p.name)) continue;
  console.log(`\n=== ${p.name} ${p.path} total ${p.requests.total} transfer ${p.requests.transfer}`);
  console.log(" byKind", Object.entries(p.requests.byKind).map(([k, v]) => `${k} ${v.requests}/${v.transfer}/${v.decoded}`).join("  "));
  const before = p.responses.filter((r) => !r.afterLoad);
  const after = p.responses.filter((r) => r.afterLoad);
  const tally = (list) => { const t = {}; for (const r of list) { const k = r.kind; t[k] ??= { n: 0, tr: 0, de: 0 }; t[k].n++; t[k].tr += r.transfer || 0; t[k].de += r.decoded || 0; } return Object.entries(t).map(([k, v]) => `${k} ${v.n}/${v.tr}/${v.de}`).join("  "); };
  console.log(" before load:", tally(before));
  console.log(" after load :", tally(after));
  if (process.env.LIST) for (const r of (process.env.LIST === "all" ? p.responses : after)) console.log("   ", r.afterLoad ? "A" : "B", r.kind.padEnd(8), String(r.status), String(r.transfer).padStart(7), String(r.decoded).padStart(7), (r.cacheControl || "").slice(0, 40).padEnd(40), r.url.slice(0, 110));
}
